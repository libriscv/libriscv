cmake_minimum_required(VERSION 3.10)
project(wasm_translation_tests CXX)
set(RISCV_32I OFF CACHE BOOL "" FORCE)
set(RISCV_64I ON CACHE BOOL "" FORCE)
set(RISCV_EXT_V OFF CACHE BOOL "" FORCE)
set(RISCV_BINARY_TRANSLATION ON CACHE BOOL "" FORCE)
set(RISCV_LIBTCC OFF CACHE BOOL "" FORCE)
set(RISCV_MEMORY_TRAPS OFF CACHE BOOL "" FORCE)
add_subdirectory(../../lib libriscv)

add_executable(difftest main.cpp)
target_link_libraries(difftest riscv)
target_compile_features(difftest PRIVATE cxx_std_20)
target_link_options(difftest PRIVATE -sNODERAWFS=1 -sALLOW_MEMORY_GROWTH=1 -sMAXIMUM_MEMORY=4GB -sEXIT_RUNTIME=1 -sSTACK_SIZE=1MB)
